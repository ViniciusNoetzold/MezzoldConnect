# ---------------- LOGIN ---------------- #

def hash_senha(senha):
    return hashlib.sha256(senha.encode()).hexdigest()

def cadastrar_usuario():
    arquivo = "usuarios.csv"

    usuario = input("Novo usuário: ").strip()
    senha = input("Senha: ").strip()

    senha_hash = hash_senha(senha)

    existe = os.path.exists(arquivo)

    with open(arquivo, "a", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)

        if not existe:
            writer.writerow(["usuario", "senha"])

        writer.writerow([usuario, senha_hash])

    print("[✓] Usuário cadastrado com sucesso")

def login():
    arquivo = "usuarios.csv"

    if not os.path.exists(arquivo):
        print("[!] Nenhum usuário cadastrado")
        cadastrar_usuario()

    usuario = input("Usuário: ").strip()
    senha = input("Senha: ").strip()

    senha_hash = hash_senha(senha)

    with open(arquivo, "r", encoding="utf-8") as f:
        reader = csv.DictReader(f)

        for row in reader:
            if row["usuario"] == usuario and row["senha"] == senha_hash:
                print("[✓] Login realizado com sucesso")
                return usuario

    print("[❌] Usuário ou senha inválidos")
    return None

def menu_login():
    while True:
        print("\n=== SISTEMA ===")
        print("1 - Login")
        print("2 - Cadastrar")
        print("3 - Sair")

        op = input("Escolha: ")

        if op == "1":
            user = login()
            if user:
                return user
        elif op == "2":
            cadastrar_usuario()
        elif op == "3":
            return None

# ---------------- NORMALIZAÇÃO ---------------- #

def normalizar_numero(numero):
    numero = numero.replace(" ", "").replace("-", "").replace("(", "").replace(")", "").replace("+", "")
    if not numero.startswith("55"):
        numero = "55" + numero
    return numero

def numero_valido(numero):
    return numero.isdigit() and numero.startswith("55") and 12 <= len(numero) <= 13

# ---------------- BLACKLIST ---------------- #

def carregar_blacklist():
    if not os.path.exists("blacklist.txt"):
        return set()
    with open("blacklist.txt", "r", encoding="utf-8") as f:
        return set(linha.strip() for linha in f)

# ---------------- CONTATOS ---------------- #

def adicionar_contatos():
    arquivo = "contatos_db.csv"
    novos = []

    print("\n[+] Adicionar contatos (FIM para sair)\n")

    while True:
        nome = input("Nome: ").strip()
        if nome.upper() == "FIM":
            break

        numero = normalizar_numero(input("Número: ").strip())

        if not numero_valido(numero):
            print("[!] Número inválido\n")
            continue

        novos.append({"nome": nome, "numero": numero})
        print(f"[✓] {nome} - {numero}")

    if not novos:
        return

    existe = os.path.exists(arquivo)

    with open(arquivo, "a", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=["nome", "numero"])
        if not existe:
            writer.writeheader()
        writer.writerows(novos)

    print(f"\n[💾] {len(novos)} contatos salvos")

def carregar_contatos(caminho):
    contatos = []
    if not os.path.exists(caminho):
        return contatos

    with open(caminho, "r", encoding="utf-8") as f:
        leitor = csv.DictReader(f)
        for linha in leitor:
            numero = normalizar_numero(linha.get("numero", ""))
            nome = linha.get("nome", "Cliente")
            contatos.append({"nome": nome, "numero": numero})

    return contatos

def remover_duplicados(contatos):
    vistos = set()
    resultado = []
    for c in contatos:
        if c["numero"] not in vistos:
            vistos.add(c["numero"])
            resultado.append(c)
    return resultado

def carregar_enviados():
    enviados = set()
    if os.path.exists("log_envios.txt"):
        with open("log_envios.txt", "r", encoding="utf-8") as f:
            for linha in f:
                enviados.add(linha.split(" - ")[1])
    return enviados

# ---------------- BUSCA ---------------- #

def buscar_contatos(contatos, ddd=None, limite=None, nome=None):
    resultado = []
    for c in contatos:
        if ddd and not c["numero"].startswith(f"55{ddd}"):
            continue
        if nome and nome.lower() not in c["nome"].lower():
            continue

        resultado.append(c)

        if limite and len(resultado) >= limite:
            break

    return resultado

# ---------------- ENVIO ---------------- #

def simular_digitacao(texto):
    print("[⌨️] Digitando...")
    for c in texto:
        sys.stdout.write(c)
        sys.stdout.flush()
        time.sleep(random.uniform(0.02, 0.05))
    print()

def enviar_midia(caminho):
    print(f"[📎] Enviando mídia: {caminho}")
    time.sleep(1)

def barra(atual, total):
    p = int((atual / total) * 20)
    print("[" + "█"*p + "-"*(20-p) + f"] {atual}/{total}")

# ---------------- LOG ---------------- #

def salvar_log(numero, status, campanha):
    with open("log_envios.txt", "a", encoding="utf-8") as f:
        data = datetime.now().strftime("%d/%m %H:%M")
        f.write(f"{data} - {numero} - {campanha} - {status}\n")

# ---------------- DELAY ---------------- #

def delay_inteligente(intervalo, contador):
    delay = intervalo + (intervalo * random.uniform(-0.3, 0.3))
    delay = max(2, delay)
    print(f"[⏳] {round(delay,2)}s...")
    time.sleep(delay)

    if contador % 10 == 0:
        pausa = random.uniform(15, 40)
        print(f"[🛑] Pausa longa {round(pausa,2)}s...")
        time.sleep(pausa)

# ---------------- EXECUÇÃO ---------------- #

def iniciar():
    campanha = input("\nNome da campanha: ")

    print("\nTipo de envio:")
    print("1 - Texto")
    print("2 - Mídia")
    print("3 - Ambos")

    tipo = input("Escolha: ")

    mensagens = []
    midia = None

    if tipo in ["1", "3"]:
        print("Digite mensagens (FIM para terminar):")
        while True:
            linha = input()
            if linha.upper() == "FIM":
                break
            mensagens.append(linha)

    if tipo in ["2", "3"]:
        midia = input("Caminho da mídia: ")
        if not os.path.exists(midia):
            print("[!] Arquivo não encontrado")
            return

    intervalo = float(input("Intervalo (segundos): "))
    arquivo = input("Arquivo de contatos: ")

    contatos = remover_duplicados(carregar_contatos(arquivo))
    blacklist = carregar_blacklist()
    enviados_antigos = carregar_enviados()

    ddd = input("DDD ou ENTER: ").strip() or None
    nome = input("Filtro nome ou ENTER: ").strip() or None
    limite = input("Limite ou ENTER: ").strip()
    limite = int(limite) if limite.isdigit() else None

    contatos = buscar_contatos(contatos, ddd, limite, nome)

    total = len(contatos)
    enviados = 0
    falhas = 0
    falhas_lista = []

    for i, c in enumerate(contatos, start=1):
        numero = c["numero"]

        if numero in blacklist:
            print(f"[🚫] Bloqueado: {numero}")
            continue

        if numero in enviados_antigos:
            print(f"[⏭️] Já enviado: {numero}")
            continue

        print(f"\n[{i}/{total}] {c['nome']} ({numero})")

        try:
            if tipo in ["1", "3"]:
                simular_digitacao(random.choice(mensagens))

            if tipo in ["2", "3"]:
                enviar_midia(midia)

            salvar_log(numero, "ENVIADO", campanha)
            enviados += 1
            print("[✓] Enviado")

        except:
            salvar_log(numero, "ERRO", campanha)
            falhas += 1
            falhas_lista.append(c)
            print("[❌] Falha")

        barra(i, total)
        delay_inteligente(intervalo, i)

    print("\n📊 RELATÓRIO FINAL")
    print(f"Campanha: {campanha}")
    print(f"Total: {total}")
    print(f"Enviados: {enviados}")
    print(f"Falhas: {falhas}")

    if total > 0:
        print(f"Taxa: {round((enviados/total)*100,2)}%")

    if falhas_lista:
        opc = input("\nDeseja reenviar falhas? (s/n): ").lower()

        if opc == "s":
            print("\n[🔁] Reenviando falhas...\n")

            for i, c in enumerate(falhas_lista, start=1):
                numero = c["numero"]

                print(f"[RETRY {i}] {c['nome']} ({numero})")

                try:
                    if tipo in ["1", "3"]:
                        simular_digitacao(random.choice(mensagens))

                    if tipo in ["2", "3"]:
                        enviar_midia(midia)

                    salvar_log(numero, "REENVIADO", campanha)
                    print("[✓] Reenviado")

                except:
                    salvar_log(numero, "FALHA FINAL", campanha)
                    print("[❌] Falhou novamente")

                time.sleep(intervalo)

# ---------------- MAIN ---------------- #

def main():
    usuario_logado = menu_login()

    if not usuario_logado:
        print("Saindo...")
        return

    while True:
        limpar_tela()
        cabecalho()
        print(f"Usuário: {usuario_logado}")
        menu()

        op = input("\nOpção: ")

        if op == "1":
            iniciar()
        elif op == "2":
            break
        elif op == "3":
            adicionar_contatos()

        input("\nENTER...")

if __name__ == "__main__":
    main()
